FILE: Swift-ios-task-main/unit 1/Palindrome and Armstrong Number.swift
================================================================================

var number = 153

// Check Palindrome
var original = number
var reverse = 0

while original > 0 {
    let digit = original % 10
    reverse = reverse * 10 + digit
    original /= 10
}

if number == reverse {
    print("Palindrome Number")
} else {
    print("Not a Palindrome Number")
}

// Check Armstrong Number
original = number
var sum = 0

while original > 0 {
    let digit = original % 10
    sum += digit * digit * digit
    original /= 10
}

if sum == number {
    print("Armstrong Number")
} else {
    print("Not an Armstrong Number")
}

================================================================================

struct BankAccount {
    var balance: Double

    mutating func deposit(amount: Double) {
        if amount > 0 {
            balance += amount
            print("Deposited: \(amount)")
        }
    }

    mutating func withdraw(amount: Double) {
        if amount > 0 && amount <= balance {
            balance -= amount
            print("Withdrawn: \(amount)")
        } else {
            print("Insufficient balance or invalid amount")
        }
    }
}

var account = BankAccount(balance: 1000)

account.deposit(amount: 500)
account.withdraw(amount: 300)

print("Final Balance: \(account.balance)")

FILE: Swift-ios-task-main/unit 3/Book and Library Classes.swift
================================================================================

class Book {
    var title: String
    var author: String

    init(title: String, author: String) {
        self.title = title
        self.author = author
    }
}

class Library {
    var books: [Book] = []

    func addBook(_ book: Book) {
        books.append(book)
        print("Book added: \(book.title)")
    }

    func removeBook(title: String) {
        if let index = books.firstIndex(where: { $0.title == title }) {
            books.remove(at: index)
            print("Book removed: \(title)")
        } else {
            print("Book not found")
        }
    }

    func searchBook(title: String) {
        if let book = books.first(where: { $0.title == title }) {
            print("Found: \(book.title) by \(book.author)")
        } else {
            print("Book not found")
        }
    }

    func listBooks() {
        print("Library Books:")
        for book in books {
            print("\(book.title) - \(book.author)")
        }
    }
}

let library = Library()

let book1 = Book(title: "Swift Basics", author: "Arun")
let book2 = Book(title: "Java Programming", author: "Kumar")

library.addBook(book1)
library.addBook(book2)

library.searchBook(title: "Swift Basics")
library.listBooks()

library.removeBook(title: "Java Programming")
library.listBooks()

FILE: Swift-ios-task-main/unit 3/Student Management System.swift
================================================================================

enum Grade: String {
    case A, B, C, D, F
}

struct Student {
    var name: String
    var rollNumber: Int
    var marks: Int

    var grade: Grade {
        switch marks {
        case 90...100: return .A
        case 80..<90: return .B
        case 70..<80: return .C
        case 60..<70: return .D
        default: return .F
        }
    }
}

class School {
    var students: [Student] = []

    func addStudent(_ student: Student) {
        students.append(student)
        print("Student added: \(student.name)")
    }

    func printReport() {
        print("\n--- Student Report ---")

        let sortedStudents = students.sorted {
            $0.marks > $1.marks
        }

        for student in sortedStudents {
            print("Name: \(student.name)")
            print("Roll Number: \(student.rollNumber)")
            print("Marks: \(student.marks)")
            print("Grade: \(student.grade.rawValue)")
            print("-------------------")
        }
    }
}

// Main Program
let school = School()

school.addStudent(Student(name: "Arun", rollNumber: 101, marks: 85))
school.addStudent(Student(name: "Kumar", rollNumber: 102, marks: 95))
school.addStudent(Student(name: "Ravi", rollNumber: 103, marks: 72))
school.addStudent(Student(name: "Priya", rollNumber: 104, marks: 55))

school.printReport()

FILE: Swift-ios-task-main/unit 3/Value Type vs Reference Type.swift
================================================================================

struct Student {
    var name: String
}

class Person {
    var name: String

    init(name: String) {
        self.name = name
    }
}

// Struct example (Value Type)
var student1 = Student(name: "Arun")
var student2 = student1

student2.name = "Kumar"

print("Original Student: \(student1.name)")
print("Copied Student: \(student2.name)")

// Class example (Reference Type)
var person1 = Person(name: "Ravi")
var person2 = person1

person2.name = "Ram"

print("Original Person: \(person1.name)")
print("Copied Person: \(person2.name)")

// Explanation:
// Struct creates an independent copy, so changing student2
// does not affect student1.
// Class copies share the same object, so changing person2
// also changes the value seen through person1.
